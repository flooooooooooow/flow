🔢 Fibonacci Sequence

Calculate Fibonacci numbers recursively

function fibonacci(n: i32) -> i32 { if n <= 1 { return n } return fibonacci(n - 1) + fibonacci(n - 2) } function main() -> i32 { print("Calculating Fibonacci numbers:") let i: i32 = 0 while i < 10 { let result: i32 = fibonacci(i) print("Fibonacci(" + i + ") = " + result) i = i + 1 } return 0 }
