const WORKGROUP_SIZE: i32 = 64;

@group(0) @binding(0) var<storage, read> a: array<f32>;
@group(0) @binding(1) var<storage, read> b: array<f32>;
@group(0) @binding(2) var<storage, read_write> out: array<f32>;

struct Params {
    n: i32,
    _pad0: u32,
    _pad1: u32,
    _pad2: u32,
};
@group(0) @binding(3) var<uniform> params: Params;

@compute @workgroup_size(64)
fn gpu_vector_add(
    @builtin(global_invocation_id) global_id: vec3<u32>,
    @builtin(workgroup_id) group_id: vec3<u32>,
    @builtin(local_invocation_id) local_id: vec3<u32>,
) {
    let tid: i32 = i32(global_id.x);

    }

Flow source

# One Flow kernel, two backends: WGSL on the browser's GPU and C on the CPU.
#
#   ./flow gpu examples/wasm/gpu_vector_add.flow      # Metal shaders
#   python3 wasm/flow_wasm_gpu.py                     # WGSL + WASM CPU reference
#
# The @gpu bodies below are the ones from lib/stdlib/gpu_kernels.flow. Two
# things read the same AST:
#
#   src/flow/wgsl_codegen.py   emits a WGSL compute shader, dispatched by the
#                              browser through WebGPU
#   src/flow/c_generator.py    emits ordinary C, compiled into the WASM module
#
# On the GPU, gpu_thread_id() is the invocation index and every element runs at
# once. On the CPU the same call reads a variable that cpu_* advances, so the
# kernel runs element by element. Same arithmetic, same order of operations,
# so the two results are comparable float for float.

extern {
    function gpu_thread_id() -> i32
    function gpu_set_thread_id(i: i32) -> void
    function sqrtf(x: f32) -> f32
    function fabsf(x: f32) -> f32
}

@gpu
function gpu_vector_add(a: ptr<f32>, b: ptr<f32>, out: ptr<f32>, n: i32) -> void {
    let i: i32 = gpu_thread_id()
    if i < n {
        out[i] = a[i] + b[i]
    }
}

@gpu
function gpu_elementwise_mul(a: ptr<f32>, b: ptr<f32>, out: ptr<f32>, n: i32) -> void {
    let i: i32 = gpu_thread_id()
    if i < n {
        out[i] = a[i] * b[i]
    }
}

@gpu
function gpu_saxpy(a: ptr<f32>, b: ptr<f32>, out: ptr<f32>, n: i32, alpha: f32) -> void {
    let i: i32 = gpu_thread_id()
    if i < n {
        out[i] = alpha * a[i] + b[i]
    }
}

# Arithmetic-heavy kernel: 128 square roots per element. The three kernels
# above are memory-bound, where a GPU cannot beat a copy; this one gives it
# something to do.
#
# The terms are independent on purpose. WGSL only promises sqrt to within 1
# ulp, so a *chained* iteration amplifies that into visible divergence: an
# earlier draft chained 256 steps and only half the elements matched, because
# the map is chaotic, not because the GPU was wrong. Summing independent terms
# keeps the disagreement at the ulp level where it belongs.
@gpu
function gpu_heavy_mix(a: ptr<f32>, b: ptr<f32>, out: ptr<f32>, n: i32) -> void {
    let i: i32 = gpu_thread_id()
    if i < n {
        let scale: f32 = b[i]
        let mut t: f32 = a[i]
        let mut acc: f32 = 0.0
        for k in 0 to 128 {
            t = t + 1.0
            acc = acc + sqrtf(fabsf(t * scale))
        }
        out[i] = acc
    }
}

# Serial CPU drivers. Each walks the same kernel body once per element.
@flow_api
export function cpu_vector_add(a: ptr<f32>, b: ptr<f32>, out: ptr<f32>, n: i32) -> void {
    for i in 0 to n {
        gpu_set_thread_id(i)
        gpu_vector_add(a, b, out, n)
    }
}

@flow_api
export function cpu_elementwise_mul(a: ptr<f32>, b: ptr<f32>, out: ptr<f32>, n: i32) -> void {
    for i in 0 to n {
        gpu_set_thread_id(i)
        gpu_elementwise_mul(a, b, out, n)
    }
}

    for i in 0 to n {
        gpu_set_thread_id(i)
        gpu_heavy_mix(a, b, out, n)
    }
}

# Fill a buffer with a deterministic, non-trivial pattern so a GPU that quietly
# returns zeros cannot pass the comparison.
@flow_api
export function fill_pattern(buf: ptr<f32>, n: i32, seed: i32) -> void {
    let mut state: i32 = seed
    for i in 0 to n {
        state = state * 1103515245 + 12345
        let bits: i32 = (state / 65536) % 2048
        buf[i] = (bits as f32) / 1024.0 - 1.0
    }
}
